<script>

(() => {

  // ---------- DOM ----------

  const phaseText = document.getElementById('phaseText');

  const timeText  = document.getElementById('timeText');

  const setPill   = document.getElementById('setPill');

  const repPill   = document.getElementById('repPill');

  const nextPill  = document.getElementById('nextPill');

  const audioState= document.getElementById('audioState');


  const startBtn  = document.getElementById('startBtn');

  const pauseBtn  = document.getElementById('pauseBtn');

  const resetBtn  = document.getElementById('resetBtn');


  const planSelect= document.getElementById('planSelect');

  const planLabel = document.getElementById('planLabel');


  const squeezeSecInput = document.getElementById('squeezeSec');

  const relaxSecInput   = document.getElementById('relaxSec');

  const repsInput       = document.getElementById('reps');

  const setsInput       = document.getElementById('sets');

  const betweenSetRestInput = document.getElementById('betweenSetRest');


  const beepToggle   = document.getElementById('beepToggle');

  const voiceToggle  = document.getElementById('voiceToggle');

  const breathToggle = document.getElementById('breathToggle');

  const vibrateToggle= document.getElementById('vibrateToggle');

  const wakeLockToggle= document.getElementById('wakeLockToggle');


  const breathCueEl  = document.getElementById('breathCue');

  const centerSmall  = document.getElementById('centerSmall');


  // ring

  const ringProgress = document.getElementById('ringProgress');

  const R = 78;

  const CIRC = 2 * Math.PI * R;


  // ---------- State ----------

  let timer = null;

  let running = false;


  let cfg = getConfigFromInputs();


  // phases: "ready" -> "squeeze" -> "relax" ... -> "between_rest" -> done

  let phase = "ready";

  let remaining = 3;      // seconds remaining in current phase

  let phaseTotal = 3;     // total seconds for current phase (for ring)

  let currentSet = 1;

  let currentRep = 1;


  // ---------- Audio ----------

  let audioCtx = null;

  let audioUnlocked = false;


  function unlockAudio() {

    if (audioUnlocked) return;

    try {

      audioCtx = audioCtx || new (window.AudioContext || window.webkitAudioContext)();

      const osc = audioCtx.createOscillator();

      const gain = audioCtx.createGain();

      osc.type = "sine";

      osc.frequency.value = 880;

      gain.gain.value = 0.0001;

      osc.connect(gain).connect(audioCtx.destination);

      osc.start();

      osc.stop(audioCtx.currentTime + 0.01);

      audioUnlocked = true;

      audioState.textContent = "🔊 声音已解锁";

    } catch (e) {

      audioState.textContent = "⚠️ 无法初始化音频";

    }

  }


  function beep(durationMs = 140, freq = 880) {

    if (!beepToggle.checked) return;

    if (!audioCtx) return;

    const t0 = audioCtx.currentTime;

    const osc = audioCtx.createOscillator();

    const gain = audioCtx.createGain();

    osc.type = "sine";

    osc.frequency.value = freq;

    gain.gain.setValueAtTime(0.0001, t0);

    gain.gain.exponentialRampToValueAtTime(0.35, t0 + 0.01);

    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + durationMs/1000);

    osc.connect(gain).connect(audioCtx.destination);

    osc.start(t0);

    osc.stop(t0 + durationMs/1000 + 0.02);

  }


  function speak(text) {

    if (!voiceToggle.checked) return;

    if (!("speechSynthesis" in window)) return;

    try {

      window.speechSynthesis.cancel();

      const u = new SpeechSynthesisUtterance(text);

      u.lang = "zh-CN";

      u.rate = 1.0;

      u.pitch = 1.0;

      window.speechSynthesis.speak(u);

    } catch (_) {}

  }


  function vibrate(pattern) {

    if (!vibrateToggle.checked) return;

    if (!navigator.vibrate) return;

    navigator.vibrate(pattern);

  }


  // ---------- Wake Lock ----------

  let wakeLock = null;

  async function setWakeLock(on) {

    if (!("wakeLock" in navigator)) return;

    try {

      if (on && !wakeLock) {

        wakeLock = await navigator.wakeLock.request("screen");

      } else if (!on && wakeLock) {

        await wakeLock.release();

        wakeLock = null;

      }

    } catch (_) {}

  }

  document.addEventListener("visibilitychange", () => {

    if (wakeLockToggle.checked && document.visibilityState === "visible") {

      setWakeLock(true);

    }

  });


  // ---------- Config ----------

  function clampInt(v, min, max){

    const n = parseInt(v, 10);

    if (Number.isNaN(n)) return min;

    return Math.max(min, Math.min(max, n));

  }

  function getConfigFromInputs() {

    const squeeze = clampInt(squeezeSecInput.value, 1, 999);

    const relax   = clampInt(relaxSecInput.value, 1, 999);

    const reps    = clampInt(repsInput.value, 1, 999);

    const sets    = clampInt(setsInput.value, 1, 99);

    const between = clampInt(betweenSetRestInput.value, 0, 999);

    return { squeeze, relax, reps, sets, between };

  }


  function applyPlan(plan) {

    if (plan === "beginner") {

      squeezeSecInput.value = 5;  relaxSecInput.value = 5;

      repsInput.value = 10;       setsInput.value = 3;

      betweenSetRestInput.value = 30;

      planLabel.textContent = "初级";

    } else if (plan === "progress") {

      squeezeSecInput.value = 8;  relaxSecInput.value = 8;

      repsInput.value = 12;       setsInput.value = 3;

      betweenSetRestInput.value = 30;

      planLabel.textContent = "进阶";

    } else if (plan === "quick") {

      squeezeSecInput.value = 1;  relaxSecInput.value = 1;

      repsInput.value = 10;       setsInput.value = 3;

      betweenSetRestInput.value = 20;

      planLabel.textContent = "快速";

    } else {

      planLabel.textContent = "自定义";

    }

    cfg = getConfigFromInputs();

    if (!running) render();

  }


  planSelect.addEventListener("change", () => applyPlan(planSelect.value));

  [squeezeSecInput, relaxSecInput, repsInput, setsInput, betweenSetRestInput].forEach(el => {

    el.addEventListener("input", () => {

      planSelect.value = "custom";

      planLabel.textContent = "自定义";

      cfg = getConfigFromInputs();

      if (!running) render();

    });

  });


  // ---------- UI helpers ----------

  function fmtSec(s){ return String(s).padStart(2, "0"); }


  function setRingProgress(p) {

    // p: 0..1

    const clamped = Math.max(0, Math.min(1, p));

    const dash = (clamped * CIRC).toFixed(2);

    ringProgress.setAttribute("stroke-dasharray", `${dash} ${CIRC.toFixed(2)}`);

  }


  function setRingColorForPhase(p){

    // 使用同一色系：S偏蓝、F偏绿、休息偏灰

    if (p === "squeeze") ringProgress.setAttribute("stroke", "rgba(110,231,255,.95)");

    else if (p === "relax") ringProgress.setAttribute("stroke", "rgba(109,255,154,.95)");

    else ringProgress.setAttribute("stroke", "rgba(255,255,255,.35)");

  }


  function setBreathCue(phase){

    // breathToggle: S=呼气，F=吸气

    if (!breathToggle.checked){

      breathCueEl.textContent = "按提示训练";

      breathCueEl.className = "breathCue muted";

      centerSmall.textContent = "保持自然呼吸，不憋气";

      return;

    }


    if (phase === "squeeze"){

      breathCueEl.textContent = "呼气";

      breathCueEl.className = "breathCue breatheOut";

      centerSmall.textContent = "轻轻S";

    } else if (phase === "relax"){

      breathCueEl.textContent = "吸气";

      breathCueEl.className = "breathCue breatheIn";

      centerSmall.textContent = "完全F";

    } else if (phase === "between_rest"){

      breathCueEl.textContent = "自然呼吸";

      breathCueEl.className = "breathCue muted";

      centerSmall.textContent = "组间休息，F肩颈";

    } else if (phase === "ready"){

      breathCueEl.textContent = "准备";

      breathCueEl.className = "breathCue muted";

      centerSmall.textContent = "就位，轻松呼吸";

    } else if (phase === "done"){

      breathCueEl.textContent = "完成";

      breathCueEl.className = "breathCue breatheIn";

      centerSmall.textContent = "很好，今日训练结束";

    } else {

      breathCueEl.textContent = "—";

      breathCueEl.className = "breathCue muted";

      centerSmall.textContent = "";

    }

  }


  // ---------- Engine ----------

  function setPhase(newPhase, sec, announceText){

    phase = newPhase;

    phaseTotal = Math.max(1, sec);

    remaining = sec;


    setRingColorForPhase(newPhase);


    // cues

    beep();

    if (announceText) speak(announceText);


    // breath voice prompt

    if (breathToggle.checked){

      if (newPhase === "squeeze") speak("呼气");

      if (newPhase === "relax") speak("吸气");

    }


    if (newPhase === "squeeze") vibrate([80, 40, 80]);

    if (newPhase === "relax") vibrate([120]);

    if (newPhase === "between_rest") vibrate([200, 80, 200]);


    render();

  }


  function nextStep(){

    if (phase === "ready") {

      currentSet = 1; currentRep = 1;

      setPhase("squeeze", cfg.squeeze, "S");

      return;

    }


    if (phase === "squeeze") {

      setPhase("relax", cfg.relax, "F");

      return;

    }


    if (phase === "relax") {

      if (currentRep < cfg.reps) {

        currentRep++;

        setPhase("squeeze", cfg.squeeze, "S");

      } else {

        if (currentSet < cfg.sets) {

          currentSet++;

          currentRep = 1;

          if (cfg.between > 0) setPhase("between_rest", cfg.between, "休息");

          else setPhase("squeeze", cfg.squeeze, "S");

        } else {

          phase = "done";

          remaining = 0;

          phaseTotal = 1;

          setRingColorForPhase("done");

          setRingProgress(1);

          beep(240, 660);

          speak("训练结束");

          vibrate([200, 100, 200, 100, 300]);

          stop();

          render();

        }

      }

      return;

    }


    if (phase === "between_rest") {

      setPhase("squeeze", cfg.squeeze, "S");

      return;

    }

  }


  function tick(){

    if (!running) return;

    remaining--;

    if (remaining <= 0) {

      nextStep();

    } else {

      if (remaining === 1) beep(90, 1040);

      render();

    }

  }


  function start(){

    unlockAudio();

    cfg = getConfigFromInputs();

    if (wakeLockToggle.checked) setWakeLock(true);


    if (phase === "done") resetState();


    if (!running) {

      running = true;

      if (phase === "ready") {

        remaining = 3;

        phaseTotal = 3;

        setRingColorForPhase("ready");

        speak("准备开始");

        beep(180, 740);

      }

      timer = setInterval(tick, 1000);

      render();

    }

  }


  function stop(){

    running = false;

    if (timer) clearInterval(timer);

    timer = null;

    if (!wakeLockToggle.checked) setWakeLock(false);

  }


  function pause(){

    if (!running) return;

    stop();

    speak("已暂停");

    render();

  }


  function resetState(){

    stop();

    cfg = getConfigFromInputs();

    phase = "ready";

    remaining = 3;

    phaseTotal = 3;

    currentSet = 1;

    currentRep = 1;

    setRingColorForPhase("ready");

    setRingProgress(0);

  }


  function reset(){

    resetState();

    speak("已重置");

    render();

  }


  function phaseLabel(p){

    if (p === "ready") return {text:"准备", cls:"rest"};

    if (p === "squeeze") return {text:"S", cls:"sq"};

    if (p === "relax") return {text:"F", cls:"fx"};

    if (p === "between_rest") return {text:"组间休息", cls:"rest"};

    if (p === "done") return {text:"完成 ✅", cls:"fx"};

    return {text:"—", cls:"rest"};

  }


  function nextLabel(){

    if (phase === "ready") return "S";

    if (phase === "squeeze") return "F";

    if (phase === "relax") {

      if (currentRep < cfg.reps) return "S";

      if (currentSet < cfg.sets) return cfg.between > 0 ? "休息" : "S";

      return "结束";

    }

    if (phase === "between_rest") return "S";

    return "—";

  }


  function render(){

    const pl = phaseLabel(phase);

    phaseText.textContent = pl.text;

    phaseText.className = "phase " + pl.cls;


    timeText.textContent = fmtSec(Math.max(0, remaining));


    setPill.textContent = `组：${Math.min(currentSet, cfg.sets)}/${cfg.sets}`;

    repPill.textContent = `次：${Math.min(currentRep, cfg.reps)}/${cfg.reps}`;

    nextPill.textContent = `下一步：${nextLabel()}`;


    // ring progress: 显示“已完成比例”

    // remaining 从 phaseTotal -> 1，进度应从 0 -> 1

    if (phase === "done") {

      setRingProgress(1);

    } else {

      const done = (phaseTotal - remaining) / phaseTotal;

      setRingProgress(done);

    }

    setBreathCue(phase);


    startBtn.textContent = running ? "进行中" : (phase === "ready" ? "开始" : "继续");

    startBtn.disabled = running;

  }


  // ---------- Events ----------

  startBtn.addEventListener("click", start);

  pauseBtn.addEventListener("click", pause);

  resetBtn.addEventListener("click", reset);


  document.getElementById("statusCard").addEventListener("click", () => unlockAudio(), {passive:true});

  wakeLockToggle.addEventListener("change", () => setWakeLock(wakeLockToggle.checked && running));


  // init

  applyPlan("custom");

  resetState();

  render();

})();

</script>
